SELECT DISTINCT p.ID, p.SEX
FROM Patient p
JOIN Laboratory l ON l.ID = p.ID
WHERE p.Admission = '+'
  AND p.Date3 BETWEEN '1990-01-01' AND '1997-12-31'
  AND l.LDH > 500
